# ----------------------------------------
#  HERO JOIN
# ----------------------------------------
# GET Hero Join
curl -X GET https://skybetbackend.bhoomi.cloud/api/joinus/herojoin

# POST Create Hero Join
curl -X POST https://skybetbackend.bhoomi.cloud/api/joinus/herojoin \
	-F "title=Hero Title" \
	-F "subtitle=Hero Subtitle" \
	-F "desp=Hero Description" \
	-F "img=@/path/to/image.jpg"

# PUT Update Hero Join
curl -X PUT https://skybetbackend.bhoomi.cloud/api/joinus/<id>/herojoin \
	-F "title=Updated Title" \
	-F "subtitle=Updated Subtitle" \
	-F "desp=Updated Description" \
	-F "img=@/path/to/image.jpg"

# DELETE Hero Join
curl -X DELETE https://skybetbackend.bhoomi.cloud/api/joinus/<id>/herojoin
