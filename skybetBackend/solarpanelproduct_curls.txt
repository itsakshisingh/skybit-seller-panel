========================================
  SOLAR PANEL PRODUCT API - CURL COMMANDS
  Base URL: https://skybetbackend.bhoomi.cloud/api/solarpanelproduct
========================================

----------------------------------------
  GET All Solar Panel Products
----------------------------------------

curl -X GET https://skybetbackend.bhoomi.cloud/api/solarpanelproduct/

----------------------------------------
  GET Solar Panel Product by ID
----------------------------------------

curl -X GET https://skybetbackend.bhoomi.cloud/api/solarpanelproduct/<id>

----------------------------------------
  POST Create Solar Panel Product
----------------------------------------

curl -X POST https://skybetbackend.bhoomi.cloud/api/solarpanelproduct/ \
  -F "name=Solar Panel 300W" \
  -F "subtitle=High Efficiency" \
  -F "percentage=21%" \
  -F "time=25 Years Warranty" \
  -F "place=Rooftop" \
  -F "description=High quality monocrystalline solar panel" \
  -F "list[]=Monocrystalline" \
  -F "list[]=High Efficiency" \
  -F "list[]=Weather Resistant" \
  -F "img=@/path/to/image.jpg"

----------------------------------------
  PUT Update Solar Panel Product
----------------------------------------

curl -X PUT https://skybetbackend.bhoomi.cloud/api/solarpanelproduct/<id> \
  -F "name=Updated Solar Panel 400W" \
  -F "subtitle=Updated Subtitle" \
  -F "percentage=22%" \
  -F "time=30 Years Warranty" \
  -F "place=Ground Mount" \
  -F "description=Updated description" \
  -F "list[]=Updated Feature One" \
  -F "list[]=Updated Feature Two" \
  -F "img=@/path/to/image.jpg"

----------------------------------------
  DELETE Solar Panel Product
----------------------------------------

curl -X DELETE https://skybetbackend.bhoomi.cloud/api/solarpanelproduct/<id>

========================================
  NOTE: Replace <id> with actual MongoDB _id of the solar panel product
========================================
