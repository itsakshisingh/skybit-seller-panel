# ========================================
#  JOB APPLICATION API - POSTMAN CURL
#  Base URL: https://skybetbackend.bhoomi.cloud/api/jobapplication
# ========================================
#  REQUIRED FIELDS:
#  - fullName (string)
#  - email (string)
#  - phone (string)
#  - department (string)
#  - experience (string)
#  OPTIONAL FIELDS:
#  - reason (string)
# ========================================

# ========================================
# CREATE JOB APPLICATION (POST)
# ========================================
curl -X POST https://skybetbackend.bhoomi.cloud/api/jobapplication \
  -H "Content-Type: application/json" \
  -d '{
    "fullName": "John Doe",
    "email": "john.doe@example.com",
    "phone": "+91 98765 43210",
    "department": "Backend Development",
    "experience": "5 years in Node.js and MongoDB",
    "reason": "I am passionate about building scalable applications"
  }'

# ========================================
# GET ALL JOB APPLICATIONS (GET)
# ========================================
curl -X GET https://skybetbackend.bhoomi.cloud/api/jobapplication

# ========================================
# GET JOB APPLICATION BY ID (GET)
# ========================================
curl -X GET https://skybetbackend.bhoomi.cloud/api/jobapplication/<id>

# ========================================
# UPDATE JOB APPLICATION (PUT)
# ========================================
curl -X PUT https://skybetbackend.bhoomi.cloud/api/jobapplication/<id> \
  -H "Content-Type: application/json" \
  -d '{
    "fullName": "Jane Doe",
    "email": "jane.doe@example.com",
    "phone": "+91 87654 32109",
    "department": "Frontend Development",
    "experience": "3 years in React and TypeScript",
    "reason": "Looking for new opportunities in innovative company"
  }'

# ========================================
# DELETE JOB APPLICATION (DELETE)
# ========================================
curl -X DELETE https://skybetbackend.bhoomi.cloud/api/jobapplication/<id>
