# ########################################
# SECTION 7: QUICK REFERENCE
# ########################################

# GET Endpoints (NO ID NEEDED):
# GET /herojoin              → Get current heroJoin
# GET /howwework             → Get current howWeWork with stats
# GET /stats                 → Get current stats
# GET /teammembers           → Get all team members
# GET /teammembers/<memberId> → Get specific team member
# GET /                      → Get all JoinUs documents
# GET /<id>                  → Get specific JoinUs document

# POST Endpoints:
# POST /                     → Create new JoinUs document
# POST /herojoin             → Create/Replace heroJoin
# POST /howwework            → Create/Replace howWeWork
# POST /teammembers          → Add new team member

# PUT Endpoints (REQUIRES ID):
# PUT /<id>                  → Update entire JoinUs document
# PUT /<id>/herojoin         → Update heroJoin
# PUT /<id>/howwework        → Update howWeWork
# PUT /<id>/stats            → Update stats
# PUT /<id>/teammembers/<memberId> → Update team member

# DELETE Endpoints (REQUIRES ID):
# DELETE /<id>               → Delete entire JoinUs document
# DELETE /<id>/herojoin      → Delete heroJoin
# DELETE /<id>/howwework     → Delete howWeWork
# DELETE /<id>/teammembers/<memberId> → Delete team member

# ########################################
# SECTION 8: IMPORTANT NOTES
# ########################################

# ID PARAMETERS:
# <id>        = JoinUs document _id (from POST / response)
# <memberId>  = Team member _id (from GET /api/joinus/teammembers response)

# CONTENT TYPES:
# -F "field=value"                              → Form data (for file uploads)
# -H "Content-Type: application/json" -d '{...}' → JSON data (for stats)

# FILE UPLOADS:
# -F "img=@/path/to/image.jpg"  → Upload image file
# Omit img field to keep existing image

# PARTIAL UPDATES:
# All fields are OPTIONAL
# Send only the fields you want to change
# Other fields remain unchanged

# RESPONSE FORMAT:
# All endpoints return JSON responses
# Check HTTP status code and response body for errors

# EXAMPLE RESPONSES:

# Success (200/201):
# {"_id": "...", "title": "...", ...}

# Error (400/404/500):
# {"message": "error description"} OR {"error": "error description"}
