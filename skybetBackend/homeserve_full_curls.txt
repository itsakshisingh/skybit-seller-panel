echo "========================================="
echo "COMPLETE HOME SERVE WORKFLOW"
echo "========================================="

# STEP 1: Create new home serve
echo ""
echo "STEP 1: Creating Home Serve..."
RESPONSE=$(curl -s -X POST https://skybetbackend.bhoomi.cloud/api/homeserve \
  -F "name=Skybet Home Services" \
  -F "img=@/path/to/service-image.jpg")
echo "$RESPONSE" | jq .
# Extract _id from response
SERVICE_ID=$(echo "$RESPONSE" | jq -r '._id')
echo "Created ServiceID: $SERVICE_ID"

# STEP 2: Get all services
echo ""
echo "STEP 2: Getting all Home Serves..."
curl -s -X GET https://skybetbackend.bhoomi.cloud/api/homeserve | jq .

# STEP 3: Get specific service
echo ""
echo "STEP 3: Getting specific Home Serve..."
curl -s -X GET https://skybetbackend.bhoomi.cloud/api/homeserve/$SERVICE_ID | jq .

# STEP 4: Update service name
echo ""
echo "STEP 4: Updating Service Name..."
curl -s -X PUT https://skybetbackend.bhoomi.cloud/api/homeserve/$SERVICE_ID \
  -F "name=Updated Skybet Services" | jq .

# STEP 5: Update service image
echo ""
echo "STEP 5: Updating Service Image..."
curl -s -X PUT https://skybetbackend.bhoomi.cloud/api/homeserve/$SERVICE_ID \
  -F "img=@/path/to/new-image.jpg" | jq .

# STEP 6: Delete service
echo ""
echo "STEP 6: Deleting Home Serve..."
curl -s -X DELETE https://skybetbackend.bhoomi.cloud/api/homeserve/$SERVICE_ID | jq .

# ########################################
# SECTION 8: QUICK REFERENCE
# ########################################

# POST /                   → Create new HomeServe (name + img required)
# GET /                    → Get all HomeServes
# GET /:id                 → Get HomeServe by ID
# PUT /:id                 → Update HomeServe (name and/or img)
# DELETE /:id              → Delete HomeServe

# ########################################
# SECTION 9: IMPORTANT NOTES
# ########################################

# REQUIRED FIELDS:
# - name (string): Service name (e.g., "Electrical Services")
# - img (file): Image file for the service

# OPTIONAL FIELDS:
# - None (all fields are required for creation)

# FILE UPLOAD:
# - Use -F "img=@/path/to/image.jpg" to upload image
# - Supported formats: jpg, png, gif, webp
# - Max file size: Check your upload middleware

# RESPONSES:
# Success (200/201):
# {"_id": "...", "name": "...", "img": "...", "createdAt": "...", "updatedAt": "..."}

# Error (400/404/500):
# {"message": "error description"} OR {"error": "error description"}

# ID PARAMETER:
# <id> = HomeServe document _id (from POST response or GET list)

# PARTIAL UPDATES (PUT):
# - Send only the fields you want to change
# - Other fields remain unchanged
# - image upload is optional if not changing image

# ########################################
# EXAMPLE: REAL WORLD SCENARIO
# ########################################
