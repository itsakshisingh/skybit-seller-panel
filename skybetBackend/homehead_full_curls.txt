# SECTION 8: QUICK REFERENCE
# ########################################

# POST /                   → Create new HomeHead (heading, category, img required)
# GET /                    → Get all HomeHeads
# GET /:id                 → Get HomeHead by ID
# PUT /:id                 → Update HomeHead
# DELETE /:id              → Delete HomeHead

# ########################################
# SECTION 9: IMPORTANT NOTES
# ########################################

# REQUIRED FIELDS:
# - heading (string): Main heading text (e.g., "Skybet Evergreen")
# - category (enum): Must be one of:
#   * SkybetEvergreen
#   * SkybetUrban
#   * SkybetIssl
#   * SkybetAlto
# - img (file): Image file for the heading

# OPTIONAL FIELDS:
# - list (array): Array of feature strings
#   Format: ["Feature 1", "Feature 2", "Feature 3"]
#   JSON string format: "[\"Feature 1\",\"Feature 2\"]"

# FILE UPLOAD:
# - Use -F "img=@/path/to/image.jpg" to upload image
# - Supported formats: jpg, png, gif, webp
# - Max file size: Check your upload middleware

# LIST FIELD:
# - Can be sent as array or JSON string
# - Example 1: -F "list=[\"feature1\",\"feature2\"]"
# - Example 2: -F "list=@/path/to/list.json"

# RESPONSES:
# Success (200/201):
# {"_id": "...", "heading": "...", "category": "...", "list": [...], "img": "...", "createdAt": "...", "updatedAt": "..."}

# Error (400):
# {"message": "Heading, img and category are required"}
# {"message": "Invalid category. Must be one of: SkybetEvergreen, SkybetUrban, SkybetIssl, SkybetAlto"}

# Error (404):
# {"message": "HomeHead not found"}
