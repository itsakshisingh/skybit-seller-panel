# ========================================
#  JOIN US API - COMPLETE UPDATED CURLS
#  Base URL: https://skybetbackend.bhoomi.cloud/api/joinus
# ========================================
#  IMPORTANT: 
#  - <id> = JoinUs document _id (from POST response)
#  - <memberId> = Team member _id (from GET /api/joinus/teammembers)
# ========================================

# ========================================
# 1. MAIN JOIN US DOCUMENT CRUD
# ========================================

# POST Create Complete JoinUs Document
curl -X POST https://skybetbackend.bhoomi.cloud/api/joinus \
  -F "heroJoin={\"title\":\"Join Our Team\",\"subtitle\":\"Be Part of Success\",\"desp\":\"Amazing opportunities awaits\"}" \
  -F "howWeWork={\"title\":\"Our Process\",\"desp\":\"Agile methodology\"}" \
  -F "heroJoinImg=@/path/to/hero.jpg" \
  -F "howWeWorkImg=@/path/to/process.jpg"

# GET All JoinUs Documents
curl -X GET https://skybetbackend.bhoomi.cloud/api/joinus

# GET JoinUs by ID (returns complete document with all nested data)
curl -X GET https://skybetbackend.bhoomi.cloud/api/joinus/<id>

# PUT Update Complete JoinUs Document
curl -X PUT https://skybetbackend.bhoomi.cloud/api/joinus/<id> \
  -F "heroJoin={\"title\":\"Updated Title\",\"subtitle\":\"Updated Subtitle\",\"desp\":\"Updated Description\"}" \
  -F "howWeWork={\"title\":\"Updated Process\",\"desp\":\"Updated Methodology\"}" \
  -F "heroJoinImg=@/path/to/new-hero.jpg" \
  -F "howWeWorkImg=@/path/to/new-process.jpg"

# DELETE JoinUs Document (deletes entire document with all nested data)
curl -X DELETE https://skybetbackend.bhoomi.cloud/api/joinus/<id>

# ========================================
# 2. HERO JOIN CRUD (NO ID NEEDED FOR GET)
# ========================================

# GET Hero Join (returns current heroJoin object - NO ID PARAMETER)
curl -X GET https://skybetbackend.bhoomi.cloud/api/joinus/herojoin

# POST Create Hero Join (creates/replaces heroJoin)
curl -X POST https://skybetbackend.bhoomi.cloud/api/joinus/herojoin \
  -F "title=Join Our Amazing Team" \
  -F "subtitle=Be Part of Success" \
  -F "desp=We are looking for talented individuals" \
  -F "img=@/path/to/hero-image.jpg"

# PUT Update Hero Join (by JoinUs document <id> - WITH ID PARAMETER)
curl -X PUT https://skybetbackend.bhoomi.cloud/api/joinus/<id>/herojoin \
  -F "title=Updated Team Title" \
  -F "subtitle=Updated Subtitle" \
  -F "desp=Updated Description" \
  -F "img=@/path/to/new-image.jpg"

# Optional: Update only specific fields (others are preserved)
curl -X PUT https://skybetbackend.bhoomi.cloud/api/joinus/<id>/herojoin \
  -F "title=Only Update Title"

# DELETE Hero Join (by JoinUs document <id>)
curl -X DELETE https://skybetbackend.bhoomi.cloud/api/joinus/<id>/herojoin

# ========================================
# 4. STATS CRUD (Nested in howWeWork)
# ========================================

# GET Stats (returns stats object from howWeWork - NO ID PARAMETER)
curl -X GET https://skybetbackend.bhoomi.cloud/api/joinus/stats

# PUT Update Stats (by JoinUs document <id> - WITH ID PARAMETER)
curl -X PUT https://skybetbackend.bhoomi.cloud/api/joinus/<id>/stats \

  -d '{
    "totalNo": 600,

# ========================================
# IMPORTANT NOTES
# ========================================
# 
# ROUTES PATTERN:
# ✅ GET /herojoin (NO ID)      → Get current heroJoin
# ✅ PUT /:id/herojoin (WITH ID) → Update by document ID
# ✅ GET /howwework (NO ID)     → Get current howWeWork
# ✅ PUT /:id/howwework (WITH ID) → Update by document ID
# ✅ GET /stats (NO ID)         → Get current stats
# ✅ PUT /:id/stats (WITH ID)   → Update by document ID
# ✅ GET /teammembers (NO ID)   → Get all team members
# ✅ PUT /:id/teammembers/:memberId (WITH BOTH IDs) → Update member
#
# IMAGE UPLOADS:
# - Use -F "img=@/path/to/image.jpg" for file upload
# - Omit img field if no image change needed
#
# FORM DATA vs JSON:
# - Form fields: curl -F "field=value"
# - JSON data: curl -H "Content-Type: application/json" -d '{...}'
#
# PARTIAL UPDATES:
# - All fields are optional
# - Send only what you want to change
# - Other fields remain unchanged
#
